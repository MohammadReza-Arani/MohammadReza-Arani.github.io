<!-- Interactive Client-side Script for Smooth Scrolling & Scrollspy -->
<script>
function scrollToCvSection(id) {
  var el = document.getElementById(id);
  if (el) {
    var offset = 80;
    var bodyRect = document.body.getBoundingClientRect().top;
    var elementRect = el.getBoundingClientRect().top;
    var elementPosition = elementRect - bodyRect;
    var offsetPosition = elementPosition - offset;
    window.scrollTo({ top: offsetPosition, behavior: 'smooth' });
  }
}

function toggleCvMobileDrawer(open) {
  var drawer = document.getElementById('cv-mobile-toc-drawer');
  var backdrop = document.getElementById('cv-mobile-toc-backdrop');
  if (drawer && backdrop) {
    if (open) {
      drawer.style.display = 'block';
      backdrop.style.display = 'block';
    } else {
      drawer.style.display = 'none';
      backdrop.style.display = 'none';
    }
  }
}

// Active TOC indicator on scroll
window.addEventListener('scroll', function() {
  var sections = ['education', 'test-scores', 'research-experience', 'teaching-experience', 'technical-skills', 'notable-coursework', 'memberships-certifications', 'download-cv'];
  var scrollPos = window.scrollY + 120;
  for (var i = sections.length - 1; i >= 0; i--) {
    var el = document.getElementById(sections[i]);
    if (el && el.offsetTop <= scrollPos) {
      var links = document.querySelectorAll('.cv-toc-link');
      links.forEach(function(l) { l.classList.remove('active'); });
      var activeLink = document.querySelector('.cv-toc-link[href="#' + sections[i] + '"]');
      if (activeLink) activeLink.classList.add('active');
      break;
    }
  }
});
</script>
